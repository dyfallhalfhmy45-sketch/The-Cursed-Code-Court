async function ask(p){ const key = $("apikey").value.trim(); if(!key) throw {code: "nofey"}; // رابط الـ API النظيف بدون أي خيارات أو عناصر زائدة const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${key}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: p }] }] }) }); const data = await res.json(); // في حال وجود خطأ من جوجل يتم عرضه بدقة if (data.error) { throw new Error(data.error.message); } if (data.candidates && data.candidates[0].content) { return data.candidates[0].content.parts[0].text; } else { throw new Error("استجابة غير صالحة من المحكمة"); } }
